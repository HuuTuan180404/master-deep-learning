class PixelEmbed(nn.Module):
    def __init__(self, img_size=224, patch_size=16, in_chans=3, in_dim=48, stride=4):
        super().__init__()
        num_patches = (img_size // patch_size) ** 2 [14*14 = 196]
        self.img_size = img_size [224]
        self.num_patches = num_patches [196]
        self.in_dim = in_dim [48]
        new_patch_size = math.ceil(patch_size / stride) [4]
        self.new_patch_size = new_patch_size [4]

        self.proj = nn.Conv2d(in_chans, self.in_dim, kernel_size=7, padding=3, stride=stride)
        # self.proj = nn.Conv2d(3, 48, kernel_size=7, padding=3, stride=4)

        self.unfold = nn.Unfold(kernel_size=new_patch_size, stride=new_patch_size)
        # self.unfold = nn.Unfold(kernel_size=4, stride=4)

    def forward(self, x, pixel_pos):
        B, C, H, W = x.shape
        assert H == self.img_size and W == self.img_size, \
            f"Input image size ({H}*{W}) doesn't match model ({self.img_size}*{self.img_size})."
        x = self.proj(x)
        x = self.unfold(x)
        x = x.transpose(1, 2).reshape(B * self.num_patches, self.in_dim, self.new_patch_size, self.new_patch_size)
        x = x + pixel_pos
        x = x.reshape(B * self.num_patches, self.in_dim, -1).transpose(1, 2)
        return x
